  2054→// ─── Main GCRM Wallet Component ──────────────────────────────────────
  2055→function GCRMWallet() {
  2056→  const currentScreen = useWalletStore((s) => s.currentScreen);
  2057→  const isWalletCreated = useWalletStore((s) => s.isWalletCreated);
  2058→  const isWalletLocked = useWalletStore((s) => s.isWalletLocked);
  2059→  const showOnboarding = useWalletStore((s) => s.showOnboarding);
  2060→  const initialize = useWalletStore((s) => s.initialize);
  2061→
  2062→  useEffect(() => {
  2063→    initialize();
  2064→  }, [initialize]);
  2065→
  2066→  const showNav = isWalletCreated && !isWalletLocked && !['create-wallet', 'import-wallet', 'seed-verify'].includes(currentScreen);
  2067→
  2068→  const renderScreen = () => {
  2069→    // Lock screen
  2070→    if (isWalletCreated && isWalletLocked) {
  2071→      return <WalletLockScreen />;
  2072→    }
  2073→
  2074→    // Onboarding
  2075→    if (showOnboarding && !isWalletCreated) {
  2076→      return <OnboardingScreen />;
  2077→    }
  2078→
  2079→    switch (currentScreen) {
  2080→      case 'create-wallet':
  2081→        return <CreateWalletScreen />;
  2082→      case 'import-wallet':
  2083→        return <ImportWalletScreen />;
  2084→      case 'dashboard':
  2085→        return <DashboardScreen />;
  2086→      case 'send':
  2087→        return <SendScreen />;
  2088→      case 'receive':
  2089→        return <ReceiveScreen />;
  2090→      case 'swap':
  2091→        return <SwapScreen />;
  2092→      case 'staking':
  2093→        return <StakingScreen />;
  2094→      case 'dapps':
  2095→        return <DAppsScreen />;
  2096→      case 'settings':
  2097→        return <SettingsScreen />;
  2098→      case 'wallet':
  2099→        return <WalletScreen />;
  2100→      case 'networks':
  2101→        return <NetworksScreen />;
  2102→      default:
  2103→        return <DashboardScreen />;
  2104→    }
  2105→  };
  2106→
  2107→  return (
  2108→    <div className="min-h-screen bg-background max-w-md lg:max-w-lg mx-auto relative">
  2109→      <ToastContainer />
  2110→      <AnimatePresence mode="wait">
